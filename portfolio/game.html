function draw() {
  background(img4);
  if (gameState=="Title"){
  titleScreen();
  } 
  if (gameState=="L1"){
  levelOne();
  } 
  if (gameState=="L2"){
   levelTwo(); 
  }
  if (gameState=="L3"){
   levelThree(); 
  }
    if (gameState=="win"){
   winGame(); 
  }
  
  text(("Score: " + score), width/2, 40);
  

} // end draw
